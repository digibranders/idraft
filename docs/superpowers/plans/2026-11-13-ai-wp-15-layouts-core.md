// upgrade.ts
/** One step per version: UPGRADE_STEPS[n - 1] turns a schema n document into schema n + 1. Empty while schema is 1. */
export type UpgradeStep = (document: Readonly<Record<string, unknown>>) => Record<string, unknown>;
export const UPGRADE_STEPS: ReadonlyArray<UpgradeStep> = [];
export const CURRENT_SCHEMA = UPGRADE_STEPS.length + 1;
/**
 * An older document raised to CURRENT_SCHEMA step by step, on read (spec 6.4). Anything else (not an object,
 * no integer schema, a newer schema, a missing step) is returned as it came, so isLayoutDocument counts it absent.
 */
export function upgradeDocument(raw: unknown, steps: ReadonlyArray<UpgradeStep> = UPGRADE_STEPS): unknown;

export const HISTORY_LIMIT = 50;

  mode: LayoutMode;
  baseline: LayoutDocument | null; // what the change count compares with: where customize started (or was rebased)
  draft: LayoutDocument | null;
  past: readonly LayoutDocument[]; future: readonly LayoutDocument[];
  focus: FocusRequest | null;
}
export type LayoutAction =
  | { type: 'enter'; document: LayoutDocument } | { type: 'exit' }
  | { type: 'edit'; next: LayoutDocument; focus?: FocusRequest | undefined }
  | { type: 'undo' } | { type: 'redo' }
  | { type: 'rebase'; baseline: LayoutDocument } | { type: 'reload'; document: LayoutDocument }
  | { type: 'focused' };

export function layoutReducer(state: LayoutState, action: LayoutAction): LayoutState;

| From | Action | To |
|---|---|---|
| view | `enter` | customize; baseline and draft are copies of `document`; history empty |
| any | `exit` | `INITIAL_LAYOUT_STATE`: draft, baseline and history end with the mode |
| customize | `enter`; `edit` with `next === draft` (a refused `draft.ts` edit) | unchanged (same object) |
| customize | `edit` | past gets the old draft (oldest dropped past 50), future cleared, draft = next, focus set |
| view | `edit`, `undo`, `redo`, `rebase`, `reload` | unchanged |
| customize | `undo` / `redo` | draft from past / future, the other stack gets the current draft; empty stack: unchanged |
| customize | `rebase` | baseline replaced, draft and history kept (team default removed while editing, spec 5.6) |
| customize | `reload` | baseline and draft = document, history cleared ("Load the other version") |
| any | `focused` | focus null |

### 2.4 `features/layouts/layout-api.ts` and `use-layout.ts` (WB-LAY-13)

```ts
export type ApiResolvedLayout = components['schemas']['ResolvedLayout'];
export const LAYOUT_PATH = '/api/v1/workspaces/{workspace_id}/layouts/{screen}';
export interface LayoutRead { layout: ApiResolvedLayout; etag: string | null }
export const layoutQueryKey = (workspaceId: string, screen: ScreenKey): QueryKey; // queryKey(LAYOUT_PATH, {...})
/** The `ETag` as `If-Match` wants it: a weak `W/` prefix (added by a compressing proxy) removed. */
export function ifMatchOf(etag: string | null): string | undefined;
export function fetchLayout(client: ApiClient, workspaceId: string, screen: ScreenKey, signal?: AbortSignal): Promise<LayoutRead>;
// putPersonalLayout and putTeamLayout(client, workspaceId, screen, document, etag): Promise<LayoutRead>;
// deletePersonalLayout and deleteTeamLayout(client, workspaceId, screen): Promise<void>.
/** The API's answer as the page uses it: built-in from the catalog, else the stored document upgraded and sanitized. */
export function toResolved(layout: ApiResolvedLayout, screen: ScreenKey): ResolvedLayoutView;
export interface ResolvedLayoutView {
  source: LayoutSource; document: LayoutDocument; teamDefaultExists: boolean;
  updatedAt: string | null; updatedBy: { userId: string; name: string } | null;

```

Each call uses `client.GET` / `PUT` / `DELETE` directly (not `unwrap`, which drops headers), reads
`response.headers.get('etag')`, and throws `problemFrom(response, error)` when `!response.ok`, so `412` arrives as an
`ApiProblem` with `code === 'precondition_failed'`. `If-Match` is sent whenever an `ETag` was read.

```ts
export interface UseLayoutResult {
  query: UseQueryResult<LayoutRead, Error>;
  savePersonal: UseMutationResult<LayoutRead, Error, { document: LayoutDocument; etag: string | null }>;
  removePersonal: UseMutationResult<void, Error, void>;
  saveTeam: UseMutationResult<LayoutRead, Error, { document: LayoutDocument; etag: string | null }>;
  removeTeam: UseMutationResult<void, Error, void>;
}
export function useLayout(workspaceId: string, screen: ScreenKey, options?: { refetch: boolean }): UseLayoutResult;
```

A `PUT` success writes its answer into the query (`setQueryData`); a `DELETE` success and any `412` invalidate the
key. `refetch: false` (customize mode) turns off window focus refetch. The contract has four writes (the spec said
"three mutations" before the team default `DELETE` existed).

Rules: `useReducer(layoutReducer)`; ids from `crypto.randomUUID().slice(0, 8)` prefixed `c-` or `n-`, checked
against the document; every action announces through `useLayoutAnnouncer` (2.8) and returns whether it changed.
`addCard` also refuses a type whose `screens` lacks the screen (spec note on WB-LAY-11: `canAddCard` first).
`save()`: 0 changes is `unchanged` with no request; a draft equal (`sameLayout`) to `defaultDocument` is a
`DELETE` of the personal layout (spec 5.6, the person keeps following the default); otherwise `PUT` with the read
`ETag`. Success: `exit`, announce `saved`. `412`: refetch, keep mode and draft, `conflict`. Other failure:
`failed`, draft kept. `saveTeamDefault()`: `PUT` team with the `ETag`, then `DELETE` personal when the source was
personal, refetch, `exit`. `removeTeamDefault()`: `DELETE` team, refetch; dirty: `rebase` on the new resolved
document, else `exit`. A refetch while customizing never touches baseline or draft.
