and `inspector.topic` (D-16).

```python
"""Reads and writes of `core.layouts` (docs/04-data-model.md section 11.6a; WP-11).

Runs in the caller's transaction as the API role. Every statement names the workspace and, for a
personal layout, the caller: that is the application half of the privacy rule (owner decision
D-09); the policy `layouts_owner` is the database half.
"""

from __future__ import annotations

from dataclasses import dataclass
from datetime import datetime
from uuid import UUID

from sqlalchemy import or_, select
from sqlalchemy.ext.asyncio import AsyncSession

from idraft.db.tables import layouts, users


@dataclass(frozen=True, slots=True)
class StoredLayout:
    id: UUID
    owner_user_id: UUID | None
    # As stored; it may not be a layout document at all (layout spec section 6.3).
    document: object
    updated_at: datetime
    updated_by: UUID
    updated_by_name: str


async def stored_layouts(
    db: AsyncSession, workspace_id: UUID, user_id: UUID, screen: str
) -> tuple[StoredLayout | None, StoredLayout | None]:
    """The caller's personal layout and the team default of the screen, in one statement."""
    rows = (
        await db.execute(
            select(
                layouts.c.id,
                layouts.c.owner_user_id,
                layouts.c.document,
                layouts.c.updated_at,
                layouts.c.updated_by,
                users.c.name,
            )
            .join(users, users.c.id == layouts.c.updated_by)
            .where(
                layouts.c.workspace_id == workspace_id,
                layouts.c.screen == screen,
                or_(layouts.c.owner_user_id == user_id, layouts.c.owner_user_id.is_(None)),
            )
        )
    ).all()
    found = [
        StoredLayout(row.id, row.owner_user_id, row.document, row.updated_at, row.updated_by, row.name)
        for row in rows
    ]
    personal = next((row for row in found if row.owner_user_id is not None), None)
    team = next((row for row in found if row.owner_user_id is None), None)
    return personal, team
```

```python
"""Layouts (docs/19-api-specification.md section 3.11.5c; layout spec sections 6.3 and 9).

What a member sees on a screen is their personal layout, else the team default, else the
built-in default the web app holds. A stored value that is not a layout document counts as
absent. The `ETag` covers the caller's personal row and the team row of the screen; a write
sends it back as `If-Match` (plan decisions L6 and L7).
"""

from __future__ import annotations

import hashlib
from contextlib import AbstractAsyncContextManager
from dataclasses import dataclass

from pydantic import ValidationError
from sqlalchemy.ext.asyncio import AsyncSession, async_sessionmaker

from idraft.api.layouts import repository
from idraft.api.layouts.catalog import SCREENS
from idraft.api.layouts.repository import StoredLayout
from idraft.api.problems import NotFoundError
from idraft.api.schemas.ai_layouts import LayoutDocument, LayoutEditor, ResolvedLayout
from idraft.api.workspaces.access import WorkspaceAccess
from idraft.db.engine import workspace_transaction


@dataclass(frozen=True, slots=True)
class Resolved:
    layout: ResolvedLayout
    etag: str


def etag(personal: StoredLayout | None, team: StoredLayout | None) -> str:
    def part(row: StoredLayout | None) -> str:
        return "-" if row is None else f"{row.id}:{row.updated_at.isoformat()}"

    return '"' + hashlib.sha256(f"{part(personal)}|{part(team)}".encode()).hexdigest()[:32] + '"'


def usable(row: StoredLayout | None) -> LayoutDocument | None:
    if row is None:
        return None
    try:
        return LayoutDocument.model_validate(row.document)
    except ValidationError:
        return None


def resolved(screen: str, personal: StoredLayout | None, team: StoredLayout | None) -> Resolved:
    team_document = usable(team)
    personal_document = usable(personal)
    if personal_document is not None:
        source, document, row = "personal", personal_document, personal
    elif team_document is not None:
        source, document, row = "team", team_document, team
    else:
        source, document, row = "builtin", None, None
    layout = ResolvedLayout.model_validate(
        {
            "screen": screen,
            "source": source,
            "document": document,
            "team_default_exists": team_document is not None,
            "updated_at": None if row is None else row.updated_at,
            "updated_by": None
            if row is None
            else LayoutEditor(user_id=row.updated_by, name=row.updated_by_name),
        }
    )
    return Resolved(layout=layout, etag=etag(personal, team))


def known(screen: str) -> None:
    if screen not in SCREENS:
        raise NotFoundError


class LayoutService:
    def __init__(self, *, sessions: async_sessionmaker[AsyncSession]) -> None:
        self._sessions = sessions

    def _transaction(self, access: WorkspaceAccess) -> AbstractAsyncContextManager[AsyncSession]:
        # The user setting is what the policy `layouts_owner` reads (owner decision D-09).
        return workspace_transaction(self._sessions, access.workspace_id, user_id=access.user_id)

    async def resolve(self, access: WorkspaceAccess, screen: str) -> Resolved:
        known(screen)
        async with self._transaction(access) as db:
            personal, team = await repository.stored_layouts(
                db, access.workspace_id, access.user_id, screen
            )
        return resolved(screen, personal, team)
```

```python
from datetime import UTC, datetime
from typing import Any

from sqlalchemy import delete, func, insert, literal, update


async def lock_screen(db: AsyncSession, workspace_id: UUID, screen: str) -> None:
    """Writers of one screen of one workspace take turns until the transaction ends, so a read,
    its If-Match comparison and the write are never overtaken (plan decision L8)."""
    await db.execute(
        select(
            func.pg_advisory_xact_lock(
                func.hashtextextended(literal(f"layouts:{workspace_id}:{screen}"), 0)
            )
        )
    )


async def insert_layout(
    db: AsyncSession,
    workspace_id: UUID,
    owner_user_id: UUID | None,
    screen: str,
    document: dict[str, Any],
    user_id: UUID,
) -> UUID:
    layout_id: UUID = (
        await db.execute(
            insert(layouts)
            .values(
                workspace_id=workspace_id,
                owner_user_id=owner_user_id,
                screen=screen,
                document=document,
                schema_version=document["schema"],
                updated_by=user_id,
            )
            .returning(layouts.c.id)
        )
    ).scalar_one()
    return layout_id


async def update_layout(
    db: AsyncSession, workspace_id: UUID, layout_id: UUID, document: dict[str, Any], user_id: UUID
) -> None:
    await db.execute(
        update(layouts)
        .where(layouts.c.workspace_id == workspace_id, layouts.c.id == layout_id)
        .values(
            document=document,
            schema_version=document["schema"],
            updated_by=user_id,
            updated_at=datetime.now(UTC),
        )
    )


async def delete_layout(
    db: AsyncSession, workspace_id: UUID, owner_user_id: UUID | None, screen: str
) -> tuple[UUID, object] | None:
    """Removes the caller's personal layout, or the team default for `owner_user_id=None`; the
    id and the document removed, or None."""
    owner = (
        layouts.c.owner_user_id.is_(None)
        if owner_user_id is None
        else layouts.c.owner_user_id == owner_user_id
    )
    row = (
        await db.execute(
            delete(layouts)
            .where(layouts.c.workspace_id == workspace_id, layouts.c.screen == screen, owner)
            .returning(layouts.c.id, layouts.c.document)
        )
    ).one_or_none()
    return None if row is None else (row.id, row.document)
```

```js
 * src/lib/layouts/catalog.ts; the web and the backend both test that the file is fresh.
 * Node strips the types of catalog.ts, which imports nothing but types.

import { createHash } from 'node:crypto';
import { readFileSync, writeFileSync } from 'node:fs';
import path from 'node:path';

import { CARD_TYPES, LIMITS, NUMBER_TYPES, SCREENS } from '../src/lib/layouts/catalog.ts';

const APP_ROOT = path.join(import.meta.dirname, '..');
const SOURCE = 'src/lib/layouts/catalog.ts';
const OUTPUT = path.join(APP_ROOT, '../../backend/src/idraft/api/layouts/catalog.json');
const byText = (a, b) => (a < b ? -1 : a > b ? 1 : 0);

const catalog = {
  generated_by: 'apps/web: pnpm gen:layout-catalog',
  source_sha256: createHash('sha256').update(readFileSync(path.join(APP_ROOT, SOURCE))).digest('hex'),
  limits: { cards: LIMITS.cards, numbers: LIMITS.numbers },
  screens: Object.fromEntries(
    Object.values(SCREENS)
      .map((screen) => [screen.key, { ...screen.zones }])
      .sort(([a], [b]) => byText(a, b)),
  ),
  // Widths keep their order: the first is the type's default.
  card_types: CARD_TYPES.map((type) => ({
    key: type.key,
    screens: [...type.screens].sort(byText),
    widths: [...type.widths],
    status: type.status,
  })).sort((a, b) => byText(a.key, b.key)),
  numbers: NUMBER_TYPES.map((number) => ({
    metric: number.metric,
    per_engine: number.perEngine,
    screens: [...number.screens].sort(byText),
  })).sort((a, b) => byText(a.metric, b.metric)),
};

const text = `${JSON.stringify(catalog, null, 2)}\n`;
if (process.argv.includes('--stdout')) process.stdout.write(text);
else writeFileSync(OUTPUT, text);
```

```python
    model_config = ConfigDict(extra="forbid", frozen=True)


class _Zones(_Strict):
    numbers: bool
    cards: bool
    widths: bool


class _CardType(_Strict):
    key: str
    screens: list[str]
    widths: list[CardWidth]
    status: Literal["active", "retired"]


class _Number(_Strict):
    metric: str
    per_engine: bool
    screens: list[str]


class _Limits(_Strict):
    cards: int
    numbers: int


class CatalogFile(_Strict):
    generated_by: str
    source_sha256: Annotated[str, StringConstraints(pattern=r"^[0-9a-f]{64}$")]
    limits: _Limits
    screens: dict[str, _Zones]
    card_types: list[_CardType]
    numbers: list[_Number]


@dataclass(frozen=True, slots=True)
class ScreenCatalog:
    numbers: bool = False
    cards: bool = False
    # Card type -> the widths it may take on this screen (none on a screen without widths).
    card_widths: Mapping[str, frozenset[str]] = field(default_factory=dict)
    # Metric -> whether it names an engine.
    metrics: Mapping[str, bool] = field(default_factory=dict)


def build(found: CatalogFile) -> dict[str, ScreenCatalog]:
    return {
        screen: ScreenCatalog(
            numbers=zones.numbers,
            cards=zones.cards,
            card_widths={
                card.key: frozenset(card.widths) if zones.widths else frozenset()
                for card in found.card_types
                if screen in card.screens
            },
            metrics={
                number.metric: number.per_engine
                for number in found.numbers
                if screen in number.screens
            },
        )
        for screen, zones in found.screens.items()
    }


FILE: Final = CatalogFile.model_validate_json(
    files(__package__).joinpath("catalog.json").read_text(encoding="utf-8")
)
CATALOG: Final = build(FILE)
_NONE: Final = ScreenCatalog()


def document_errors(
    screen: str, document: LayoutDocument, catalog: Mapping[str, ScreenCatalog] = CATALOG
) -> list[FieldError]:
    allowed = catalog.get(screen, _NONE)
    found: list[FieldError] = []
    if document.numbers and not allowed.numbers:
        found.append(FieldError("numbers", "not_on_screen", "This screen has no headline numbers."))
    else:
        for index, number in enumerate(document.numbers or []):
            per_engine = allowed.metrics.get(number.metric)
            at = f"numbers.{index}"
            if per_engine is None:
                found.append(FieldError(f"{at}.metric", "not_in_catalog", "Not a number of this screen."))
            elif per_engine and number.engine is None:
                found.append(FieldError(f"{at}.engine", "required", "An AI number names its engine."))
            elif not per_engine and number.engine is not None:
                found.append(FieldError(f"{at}.engine", "not_applicable", "Only AI numbers name an engine."))
    if document.cards and not allowed.cards:
        found.append(FieldError("cards", "not_on_screen", "This screen has no cards."))
    else:
        for index, card in enumerate(document.cards or []):
            widths = allowed.card_widths.get(card.type)
            if widths is None:
                found.append(FieldError(f"cards.{index}.type", "not_in_catalog", "Not a card of this screen."))
            elif card.width is not None and card.width not in widths:
                found.append(FieldError(f"cards.{index}.width", "width_not_allowed", "Not a width this card takes here."))
    return found


def document_size(document: Mapping[str, Any]) -> int:
    """Bytes of the document as PostgreSQL prints a jsonb value: the same separators as
    `json.dumps`, non-ASCII as UTF-8, so the API and the table's check measure the same text."""
    return len(json.dumps(document, ensure_ascii=False).encode())


def stored_document(screen: str, body: LayoutDocument) -> dict[str, Any]:
    """The document as stored, or `422 validation_failed` naming every field found wrong."""
    document = body.model_dump(mode="json", by_alias=True, exclude_none=True)
    if document_size(document) > DOCUMENT_MAX_BYTES:
        raise invalid_fields([FieldError("", "too_large", "A layout is at most 16 KB.")])
    found = document_errors(screen, body)
    if found:
        raise invalid_fields(found)
    return document
```
